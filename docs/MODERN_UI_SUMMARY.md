### Basic Implementation
```html
<!-- Include CSS -->
<link rel="stylesheet" href="./css/modern-styles.css">
<link rel="stylesheet" href="./css/animations.css">

<!-- Include JS -->
<script src="./js/modern-ui.js"></script>

<!-- Use components -->
<button class="btn btn--gold">Win Big!</button>
<div class="card neon-border">Card with neon</div>
```

### JavaScript API
```javascript
// Toast notifications
const toast = new ToastManager();
toast.success('Success!');

// Winner celebration
showWinnerAnnouncement(50000, 'John');

// Confetti effect
new ConfettiEffect().create(3000);

// Countdown timer
new CountdownTimer('2024-12-31', 'timerId');

// Particle burst
new ParticleSystem('particles').burst(x, y);
```
